local unpackValues=table.unpack or unpack

local function section(s,a,b)local x=assert(s:find(a,1,true),a);return s:sub(x,assert(s:find(b,x+#a,true),b)-1)end
local function loadenv(body,label,env)
 if _VERSION=='Lua 5.1'then local fn,err=loadstring(body,label);if fn then setfenv(fn,env)end;return fn,err end
 return load(body,label,'t',env)
end
local function pack(...)return{n=select('#',...),...}end
M.read=read;M.section=section;M.loadenv=loadenv
function M.compileSource(source,label,petSource)
 local geometry=section(source,'local function ValueIsSecret(value)','local function PlayerDeadOrGhost(unit)')
 local empty=section(source,'local function HideEmptyUnitPortrait(self)','local function ApplyWatchedUnitChrome')

 local atlas=section(source,'function UF.UpdateAtlasBorder(self)','-- Unit tooltip (custom SecureUnitButton')

 local body='local _,ns=...;local UF,API=ns.UnitFrames,ns.API\n'..geometry..empty..portraits..atlas..partition
  ..'\nreturn ApplyPortraitCamera,SchedulePortraitCameraRefresh,SetPortraitUnit'
 -- Compile once outside any later cost window. Environment is applied per fixture.
 local pet=petSource or read('UnitFrames/FluxUnitFrames_Pet.lua')
 local petBody=section(pet,'    local function UpdatePetData()','    ns.PetFrame = frame')
 return{path=label,body=body,petBody=petBody,source=source}
end
function M.compile(path,petPath)
 return M.compileSource(read(path),path,petPath and read(petPath))
end
