local checks,expectedFailures=0,0
local function check(v,label)checks=checks+1;assert(v,label)end
local function read(path)local f=assert(io.open(path,"rb"));local s=f:read("*a"):gsub("\r\n","\n");f:close();return s end
local function part(path,first,last)
 local text=read(path);local a=assert(text:find(first,1,true),first);local b=last and assert(text:find(last,a+#first,true),last) or (#text+1)
 return text:sub(a,b-1),select(2,text:sub(1,a-1):gsub("\n",""))
end
local opaque=setmetatable({}, {
 __lt=function()error("SIMULATED_SECRET_ALPHA_ORDER",2)end,
 __le=function()error("SIMULATED_SECRET_ALPHA_ORDER",2)end,
 __div=function()error("SIMULATED_SECRET_ALPHA_DIVISION",2)end,
 __sub=function()error("SIMULATED_SECRET_ALPHA_SUBTRACTION",2)end,
})
local function environment(values)
 local e=setmetatable(values or {},{__index=_G});e._G=e
 e.type=function(v)if rawequal(v,opaque)then return "number"end;return type(v)end
 e.issecretvalue=function(v)return rawequal(v,opaque)end
 e.canaccessvalue=function(v)return not rawequal(v,opaque)end
 return e
end
local function compile(path,first,last,suffix,e)
 local source,offset=part(path,first,last)
 return assert(load(string.rep("\n",offset)..source.."\n"..(suffix or ""),"@"..path,"t",e))()
end
local function frame(alpha)
 local f={alpha=alpha,shown=true,reads=0}
 function f:GetAlpha()self.reads=self.reads+1;return self.alpha end
 function f:GetEffectiveAlpha()self.reads=self.reads+1;return self.alpha end
 function f:SetAlpha(v)self.alpha=v end
 function f:IsShown()return self.shown end
 function f:GetTexture()return "art"end
 function f:SetDrawBling(v)self.bling=v end
 function f:SetDrawEdge(v)self.edge=v end
 function f:SetDrawSwipe(v)self.swipe=v end
 function f:SetSwipeColor()end
 function f:SetCooldownFromDurationObject()end
 function f:Hide()self.shown=false end
 return f
end

-- CastBarIsBusy alpha checks are return-redundant. Remove them only in an
-- in-memory candidate and compare all public controls; no production file edit.
local busySource=part("UnitFrames/FluxSwingTimer.lua","local function CastBarIsBusy()","local function SwingTimerFeatureWanted()")
local guardStart=assert(busySource:find("    -- Only block when the cast bar is actually on-screen.",1,true))
local candidate=busySource:sub(1,guardStart-1).."    return false\nend\nreturn CastBarIsBusy"
for _,shown in ipairs({false,true})do
 for _,alpha in ipairs({0,.005,.01,.4,.75,1})do
  for flag=0,6 do
   local bar=frame(alpha);bar.shown=shown
   local field=({"casting","channeling","customTimer","preInterrupt","holdTime","fadeOut"})[flag]
   if field then bar[field]=field=="holdTime" and .2 or true end
   local e=environment({castBar=bar})
   local original=assert(load(busySource.."\nreturn CastBarIsBusy","@busy-original","t",e))()
   local improved=assert(load(candidate,"@busy-public-state-experiment","t",e))()
   check(original()==improved(),"busy public control retains exact result")
  end
 end
end
local bar=frame(opaque);local e=environment({castBar=bar})
local improved=assert(load(candidate,"@busy-public-state-experiment","t",e))()
check(improved()==false and bar.reads==0,"busy candidate preserves no-active-cast result without alpha read")
