local checks,expectedFailures,containedFailures=0,0,0
local function check(ok,label)checks=checks+1;assert(ok,label)end
local function read(path)local f=assert(io.open(path,"rb"));local s=f:read("*a"):gsub("\r\n","\n");f:close();return s end
local function fragment(path,first,last)
    local source=read(path);local a=assert(source:find(first,1,true),first)
    local b=assert(source:find(last,a+#first,true),last)
    local _,lines=source:sub(1,a-1):gsub("\n","")
    return string.rep("\n",lines)..source:sub(a,b-1)
end
local function loadfragment(e,path,first,last,export)
    return assert(load(fragment(path,first,last).."\nreturn "..export,"@"..path,"t",e))()
end
local function environment()
    local e=setmetatable({},{__index=_G});e._G=e
    local s={traps=0,alphaSets=0}
    local function trap(kind)
        return function()s.traps=s.traps+1;error("MODELED_OPAQUE_"..kind,2)end
    end
    s.opaque=setmetatable({opaque=true},{__lt=trap("COMPARE"),__le=trap("COMPARE"),__add=trap("ARITHMETIC"),__sub=trap("ARITHMETIC"),__mul=trap("ARITHMETIC"),__div=trap("ARITHMETIC")})
    e.issecretvalue=function(value)return type(value)=="table"and rawget(value,"opaque")==true end
    e.canaccessvalue=function()return true end -- Deliberately true even for opaque.
    local ns={Flux={Client={isRetail=true,isCamelot=true}},API={},_FluxAPICompat={issecretvalue=e.issecretvalue}}
    e.FLUX_API={};e.WOW_VERSION={}
    assert(loadfile("Core/API/Secrets.lua","t",e))("_FluxUI",ns)
    assert(loadfile("Core/API/Secrets_Midnight.lua","t",e))("_FluxUI",ns)
    assert(loadfile("Core/FluxAPI/Secrets.lua","t",e))("_FluxUI",ns)
    e.API=ns.API;e.UF={};s.ns=ns;s.env=e
    return s,e
end

-- Actual secret helpers reject an opaque value even though CanAccessValue=true.
do
    local s,e=environment();local api=s.ns.API;local sec=s.ns.Flux.Secrets
    check(api.CanAccessValue(s.opaque)==true,"model deliberately supplies accessible-but-secret value")
    check(api.IsSecretValue(s.opaque)==true,"production legacy helper identifies opaque stand-in")
    check(api.SafeGetNumber(s.opaque,7)==7 and sec.SafeGetNumber(s.opaque,8)==8,"both production number helpers reject secret before conversion")
    check(api.KeepLastNumber(s.opaque,.4,1)==.4,"production KeepLastNumber retains prior plain alpha")
    check(api.SafeGetNumber(.7,0)==.7 and api.SafeGetNumber(nil,.9)==.9,"readable/missing numeric helper controls")
    check(s.traps==0,"guarded helpers never compare or calculate opaque inputs")
    print("RESULT|secret_helper_controls|checks=6|opaque_operations=0|canaccessvalue_true_is_not_enough=true")
end

-- Real tiled-overlay helper: arithmetic failure is caught internally, leaving
-- previous coordinates and clearing its public geometry cache for the attempt.
do
    local s,e=environment()
    e.ValueIsSecret=loadfragment(e,shared,"local function ValueIsSecret(value)","local function PlayerDeadOrGhost(unit)","ValueIsSecret")
    local refresh=loadfragment(e,shared,"local function RefreshTiledOverlayTexCoords(overlay, tileSize)","local function RefreshAbsorbShieldOverlay(bar, tileSize)","RefreshTiledOverlayTexCoords")
    local o={width=96,height=32,writes=0,shown=true}
    function o:IsShown()return self.shown end
    function o:GetWidth()return self.width end
    function o:GetHeight()return self.height end
    function o:SetTexCoord(...)self.coords={...};self.writes=self.writes+1 end
    refresh(o,32);check(o.writes==1 and o.coords[2]==3 and o.coords[4]==1,"plain geometry computes correct tile coordinates")
    refresh(o,32);check(o.writes==1,"stable plain geometry skips duplicate TexCoord write")
    local prior=o.coords;o.width=s.opaque
    check(pcall(refresh,o,32),"opaque geometry error is contained by production pcall")
    check(s.traps==1 and o.writes==1 and o.coords==prior,"contained opaque division leaves previous tile coordinates")
    check(o._fluxTileWidth==nil and o._fluxTileHeight==nil and o._fluxTileSize==nil,"opaque attempt clears public dimension caches")
    containedFailures=containedFailures+1
    o.width=64;refresh(o,32);check(o.writes==2 and o.coords[2]==2,"later readable geometry recovers")
    o.shown=false;o.width=s.opaque;local before=s.traps;refresh(o,32)
    check(s.traps==before,"hidden overlay avoids geometry reads/arithmetic")
    check(pcall(refresh,nil,32),"missing overlay exits safely")
    print("RESULT|tiled_overlay_geometry|plain_control=PASS|stable_cache_control=PASS|hidden_control=PASS|missing_control=PASS|contained_conditional_arithmetic_failure=1|subsequent_public_recovery=PASS")
end
