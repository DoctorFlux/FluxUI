local function read(path)
    local f=assert(io.open(path,"rb"));local s=f:read("*a");f:close();return s:gsub("\r\n","\n")
end
local function cut(s,first,last)
    local a=assert(s:find(first,1,true),first);return s:sub(a,assert(s:find(last,a+#first,1,true),last)-1)
end

-- Execute the COMPLETE current usability module. Only the native option parser
-- and underlying public usability results are modeled; no secure macro executes.
local usability=read("ExtraButtonSystem/Engine/Usability.lua")
local macroFailures,macroPairs,parserCalls=0,0,0
for _,client in ipairs(clients)do
    local e=env();e.FLUX_API={isLegacyClient=client:find("classic",1,true)~=nil}
    local calls=0;local active="ActiveBlocked"
    e.SecureCmdOptionParse=function(tail)calls=calls+1;return active end
    e.GetInventoryItemID=function(_,slot)return 10000+slot end
    assert(load(usability,"@ExtraButtonSystem/Engine/Usability.lua","t",e))()
    local u=e.FluxButtonUsability
    u.CheckSpellUsability=function(_,spell)return spell=="ActiveBlocked"and u.STATE_NOT_USABLE or u.STATE_USABLE,false end
    u.CheckItemUsability=function(_,item)return item==10013 and u.STATE_NOT_USABLE or u.STATE_USABLE,false end
    local pe=env();pe.FLUX_API=e.FLUX_API;pe.SecureCmdOptionParse=e.SecureCmdOptionParse;pe.GetInventoryItemID=e.GetInventoryItemID
    -- Isolated proposed typo correction only, held in memory; no addon changes.
    local proposed,n=usability:gsub('find%("%%%[", 1, true%)','find("[", 1, true)')
    check(n==2,"exactly two plain-search bracket typos found")
    assert(load(proposed,"@proposal-only-usability","t",pe))()
    local p=pe.FluxButtonUsability;p.CheckSpellUsability=u.CheckSpellUsability;p.CheckItemUsability=u.CheckItemUsability
    for i=1,2000 do
        active=i%2==0 and"ActiveBlocked"or"ActiveReady"
        local before=calls
        local actual=u:CheckMacroUsability("/cast [mod:shift,@mouseover,help] ActiveBlocked; ActiveReady","spell",111)
        check(calls==before,"current conditional path makes no parser call")
        local corrected=p:CheckMacroUsability("/cast [mod:shift,@mouseover,help] ActiveBlocked; ActiveReady","spell",111)
        local expected=active=="ActiveBlocked"and u.STATE_NOT_USABLE or u.STATE_USABLE
        check(corrected==expected,"isolated typo correction selects active public clause")
        if actual~=expected then macroFailures=macroFailures+1 end
        macroPairs=macroPairs+1
    end
    active="13"
    local actual=u:CheckMacroUsability("/use [mod:shift] 13; 14",nil,nil,nil)
    local corrected=p:CheckMacroUsability("/use [mod:shift] 13; 14",nil,nil,nil)
    check(actual==u.STATE_USABLE and corrected==u.STATE_NOT_USABLE,"conditional inventory-slot use incorrectly defaults usable")
    active=nil
    check(p:CheckMacroUsability("/cast [@mouseover,help] Missing",nil,nil,nil)==u.STATE_USABLE,"no matching clause retains documented neutral peek fallback")
    check(p:CheckMacroUsability("/cast PlainSpell","spell",111)==u.STATE_USABLE,"plain cast metadata fallback retained")
    parserCalls=parserCalls+calls
end
findings.macro={id="R2-EBS-01",classification="dormant legacy helper; no demonstrated current TOC caller",pairs=macroPairs,mismatches=macroFailures,inventorySlotMismatches=#clients,proposalOnlyParserCalls=parserCalls}

-- COMPLETE Charges module. Setter failure is explicitly conditional injection,
-- not a claim that a native public SetCooldown currently fails on any client.
local chargeSource=read("Core/ButtonFX/Charges.lua")
local function chargesBoot(failOnce)
    local e=env();local ns={Flux={Client={isAnyClassic=false},Secrets={IsSecretValue=function()return false end,
        CanAccessValue=function()return true end,SafeGetNumber=function(v,d)return tonumber(v)or d end}}}
    local cd={attempts=0,successes=0,shown=true,duration=0}
    function cd:SetCooldown(start,duration)self.attempts=self.attempts+1;if failOnce and self.attempts==1 then error("INJECTED_TRANSIENT_SETTER_FAILURE")end;self.start,self.duration=start,duration;self.successes=self.successes+1 end
    function cd:GetEffectiveAlpha()return 1 end
    function cd:SetDrawBling(v)self.bling=v end
    function cd:SetDrawEdge(v)self.edge=v end
    function cd:SetDrawSwipe(v)self.swipe=v end
    function cd:SetSwipeColor()end
    function cd:Show()self.shown=true end
    function cd:Hide()self.shown=false end
    function cd:IsShown()return self.shown end
    function cd:Clear()self.duration=0 end
    assert(load(chargeSource,"@Core/ButtonFX/Charges.lua","t",e))("_FluxUI",ns)
    return ns.Flux.ButtonFX.Charges,{cooldown=cd},cd
end
local opts={start=100,duration=8,enable=1,skipDurationObject=true,chargeIsActive=false,hideWhenIdle=true}
local fx,w,cd=chargesBoot(true)
for i=1,1000 do fx.Apply(w,opts)end
check(cd.attempts==1 and cd.successes==0 and cd.duration==0 and cd._fxSigShown==true,"failed numeric native setter cached as successfully shown")
local failed={attempts=cd.attempts,successes=cd.successes,cachedShown=cd._fxSigShown,duration=cd.duration}
fx.ClearCache(w);fx.Apply(w,opts)
check(cd.attempts==2 and cd.successes==1 and cd.duration==8,"cache invalidation control allows same timer to recover")
local good,gw,gcd=chargesBoot(false)
for i=1,1000 do good.Apply(gw,opts)end
check(gcd.attempts==1 and gcd.successes==1 and gcd.duration==8,"normal unchanged timer retains one native write budget")
findings.failedSetter={id="R2-EBS-04",classification="conditional setter-failure model",failed=failed,
    recoveryAfterExplicitClear={attempts=cd.attempts,successes=cd.successes},publicControlWrites=gcd.attempts,applyCalls=2002}

local function json(v)
    if type(v)=="table"then local keys={};for k in pairs(v)do keys[#keys+1]=k end;table.sort(keys,function(a,b)return tostring(a)<tostring(b)end);local out={};for _,k in ipairs(keys)do out[#out+1]=json(tostring(k))..":"..json(v[k])end;return"{"..table.concat(out,",").."}"
    elseif type(v)=="string"then return string.format("%q",v):gsub("\n","n")
    elseif type(v)=="nil"then return"null"else return tostring(v)end
end
