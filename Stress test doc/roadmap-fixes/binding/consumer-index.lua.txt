-- Actual Bars State/slot bookkeeping/Watch modifier callback and full EBS
-- HotkeyUtil. Counters are desktop call counts, not native CPU/FPS/taint proof.
local sourceRoot=arg[1]or"."
local function read(path)local f=assert(io.open(sourceRoot.."/"..path,"rb"));local s=f:read("*a"):gsub("\r\n","\n");f:close();return s end
local function cut(s,a,b)local p=assert(s:find(a,1,true),a);return s:sub(p,assert(s:find(b,p+#a,true),b)-1)end
local stateSource=read("Bars/State.lua");local watchSource=read("Bars/Watch.lua");local hostSource=read("Bars/Hosts.lua")
local checks,rawEvents=0,0;local function check(v,why)checks=checks+1;assert(v,why)end
local clients={"classic_era","classic_anniversary","live","forever"}
local function barsFixture(client,combat,count,supported)
    local s={combat=false,mask=0,hostVisits=0,walks=0,reads=0,writes=0,removes=0,paintRequests=0,paintTokens={},dirty=0,badge=0,effects=0}
    local e=setmetatable({},{__index=_G});e._G=e
    e.IsShiftKeyDown=function()return s.mask%2==1 end;e.IsControlKeyDown=function()return math.floor(s.mask/2)%2==1 end
    e.IsAltKeyDown=function()return math.floor(s.mask/4)%2==1 end
    e.table={};for k,v in pairs(table)do e.table[k]=v end
    e.table.remove=function(t,i)s.removes=s.removes+1;return table.remove(t,i)end
    local hosts={};local bars={};local flux={Bars=bars,Combat={InLockdown=function()return s.combat end},
        Restricted={PagingWorks=function()return supported end,Works=function()return client~="forever"end},
        Context={HasSubscribers=function()return true end}}
    e.Flux=flux;local ns={Flux=flux}
    assert(load(read("Bars/SlotMath.lua"),"@production-slot-math","t",e))("_FluxUI",ns)
    bars.Hosts={};local index=assert(load("local Hosts=...;local slotIndex={}\n"
        ..cut(hostSource,"function Hosts.IndexSlot(host, newSlot)","function Hosts.ForSlot(slot)").."\nreturn slotIndex","@production-index-slot","t",e))(bars.Hosts)
    bars.Paint={Dirty=function()s.dirty=s.dirty+1 end};bars.BadgeHost={NotifyDisplay=function()s.badge=s.badge+1 end}
    bars.Effects={Refresh=function()s.effects=s.effects+1 end}
    bars.ForEachHost=function(fn)s.walks=s.walks+1;for _,host in ipairs(hosts)do s.hostVisits=s.hostVisits+1;fn(host)end end
    bars.RequestPaint=function(key)s.paintRequests=s.paintRequests+1;s.paintTokens[key]=true end
    for i=1,count do
        local col=(i-1)%19+1;local row=({"main","sec","tert"})[math.floor((i-1)/19)+1]
        local widget={attrs={["flux-column"]=col,["flux-has-sec"]=1,["flux-has-tert"]=1,["flux-promoted"]=0}}
        function widget:GetAttribute(k)s.reads=s.reads+1;return self.attrs[k]end
        function widget:SetAttribute(k,v)if s.combat then error("protected combat attribute write")end;s.writes=s.writes+1;self.attrs[k]=v end
        hosts[#hosts+1]={widget=widget,column=col,row=row}
    end
    assert(load(stateSource,"@production-bars-state","t",e))("_FluxUI",ns)
    bars.State.ApplySlotsInsecure() -- Normal out-of-combat preparation.
    local ev={};ev.Register=function(name,callback)s.event=callback end
    local callbackCode="local Flux,Bars,Ev=...;local InLockdown=function()return Flux.Combat.InLockdown()end\n"
        ..cut(watchSource,"\tlocal function ContextOwnsDisplay()","\t-- Extra Watch used to be the only")
        ..cut(watchSource,'\tEv.Register("MODIFIER_STATE_CHANGED", function()','\tEv.Register("UPDATE_BINDINGS", function()')
    assert(load(callbackCode,"@production-bars-watch-modifier","t",e))(flux,bars,ev)
    s.combat=combat;s.hosts,s.index,s.bars=hosts,index,bars
    for _,name in ipairs({"hostVisits","walks","reads","writes","removes","paintRequests","dirty","badge","effects"})do s[name]=0 end
    return s
end

local visits,mutations,moves=0,0,0
for _,target in ipairs({{'classic_era',true},{'classic_anniversary',true},{'live',true},{'forever',true},{'forever',false}})do
 for _,combat in ipairs({false,true})do for _,count in ipairs({0,1,57})do
  for _,pattern in ipairs({'duplicate-held','gray-chords'})do
   local s=barsFixture(target[1],combat,count,target[2]);local logical=0;local gray={0,1,3,2,6,7,5,4};local edgeCount=1024
   for edge=1,edgeCount do
    local before={};for i,host in ipairs(s.hosts)do before[i]=host.slot end
    s.mask=pattern=='duplicate-held'and 1 or gray[(edge-1)%8+1]
    s.event('MODIFIER_STATE_CHANGED','LSHIFT',edge%2)
    for i,host in ipairs(s.hosts)do if host.slot~=before[i]then logical=logical+1 end end
   end
   local expectedVisits=(combat or not target[2])and count*edgeCount or 0
   check(s.hostVisits==expectedVisits,'supported paging capability and immediate index timing preserved')
   check(s.removes==logical,'only changed memberships are mutated: removes='..s.removes..' moves='..logical)
   if combat then
    check(s.writes==0,'combat indexes without protected attributes')
    check(s.dirty==logical and s.badge==logical and s.effects==logical,'each real move retains paint/badge/effect cadence')
   elseif not target[2]then
    check(s.writes==count*edgeCount*26,'unsupported paging retains existing OOC fallback contract')
   else check(s.writes==0,'supported paging modifier edge does not restamp attributes')end
   local indexed=0;for _,list in pairs(s.index)do for _ in ipairs(list)do indexed=indexed+1 end end
   check(indexed==count,'reverse index membership stays exact')
   local buckets=0;for _ in pairs(s.paintTokens)do buckets=buckets+1 end
   check(buckets==4,'downstream paint buckets still coalesce independently')
   if count==57 then print(('CONSUMER|%s|paging=%s|combat=%s|pattern=%s|edges=%d|visits=%d|removes=%d|logical_moves=%d|attribute_writes=%d'):format(target[1],tostring(target[2]),tostring(combat),pattern,edgeCount,s.hostVisits,s.removes,logical,s.writes))end
   visits=visits+s.hostVisits;mutations=mutations+s.removes;moves=moves+logical
  end
 end end
end
print(('CONSUMER ACCEPTANCE|checks=%d|host_visits=%d|membership_mutations=%d|logical_moves=%d|native=not-run'):format(checks,visits,mutations,moves))
