-- Standalone test-only path overlay. Never loaded by WoW or copied into AddOns.
local baseline,mode,script,live=assert(arg[1]),assert(arg[2]),assert(arg[3]),assert(arg[4])
local roots={};for i=5,#arg do roots[#roots+1]=arg[i]end
local function norm(s)return(s:gsub('\\','/'):gsub('/+$',''))end
local function clean(s)
 local parts={}
 for part in s:gmatch('[^/]+')do
  if part=='..'and#parts>1 then table.remove(parts)
  elseif part~='.'then parts[#parts+1]=part end
 end
 return table.concat(parts,'/')
end
baseline,live=norm(baseline),norm(live)
for i,p in ipairs(roots)do roots[i]=norm(p)end
local originalOpen,originalLoadfile=io.open,loadfile
local cache={}
local function exists(p)local f=originalOpen(p,'rb');if not f then return false end;f:close();return true end
local function resolve(path)
 if type(path)~='string'then return path end
 path=norm(path)
 if path:sub(1,2)=='./'or path:sub(1,3)=='../'then path=clean(live..'/'..path)end
 local relative=path
 if path:lower():sub(1,#live+1)==live:lower()..'/'then relative=path:sub(#live+2)
 elseif path:lower():sub(1,#baseline+1)==baseline:lower()..'/'then relative=path:sub(#baseline+2)
 elseif path:match('^%a:/')or path:sub(1,1)=='/'or path:sub(1,3)=='../'then return path end
 if cache[relative]then return cache[relative]end
 if mode=='candidate'then
  for _,root in ipairs(roots)do
   local p=root..'/'..relative
   if exists(p)then cache[relative]=p;return p end
  end
 end
 local p=baseline..'/'..relative
 if exists(p)then cache[relative]=p;return p end
 -- Assets and supplied external native/reference fixtures remain read-only.
 return path
end
io.open=function(path,access)
 access=access or'r'
 if access:find('[wa+]')then error('PROPOSAL_TEST_NO_FILESYSTEM_WRITES',2)end
 return originalOpen(resolve(path),access)
end
loadfile=function(path,access,environment)
 local f,e=originalOpen(resolve(path),'rb');if not f then return nil,e end
 local source=f:read('*a');f:close()
 source=source:gsub('^\239\187\191',''):gsub('^#![^\n]*','')
 -- Preserve logical chunk names used by unchanged debug-hook work counters.
 return load(source,'@'..path,access or'bt',environment or _ENV)
end
dofile=function(path)local f,e=loadfile(path);if not f then error(e,2)end;return f()end
arg={[0]=script}
local ok,err=xpcall(function()dofile(script)end,debug.traceback)
if not ok then io.stderr:write(err..'\n');os.exit(1)end
